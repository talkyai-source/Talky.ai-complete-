TENANT = '20000000-0000-4000-8000-000000000017'
LEAD = '30000000-0000-4000-8000-000000000017'
PHONE = '+15555550107'


class Connection:
    def __init__(self, expired_source=None):
        self.lead = {'id': LEAD, 'status': 'calling', 'last_call_result': None,
                     'do_not_call': False, 'call_attempts': 0}
        self.entry = self.make_entry(expired_source, datetime.now(timezone.utc) - timedelta(days=1)) if expired_source else None

    def make_entry(self, source, expiry):
        return {'id': '40000000-0000-4000-8000-000000000017', 'tenant_id': TENANT,
                'normalized_number': PHONE, 'source': source, 'reason': 'synthetic',
                'expires_at': expiry, 'created_at': datetime.now(timezone.utc)}

    async def __aenter__(self):
        return self

    async def __aexit__(self, *args):
        return False

    def transaction(self):
        return self

    async def execute(self, sql, *args):
        if sql.strip().startswith('SET LOCAL'):
            return 'SET'
        if 'UPDATE dialer_jobs' in sql:
            return 'UPDATE 0'
        if 'UPDATE leads' in sql:
            self.lead.update(status=args[1], last_call_result=args[2], call_attempts=args[3])
            return 'UPDATE 1'
        if 'INSERT INTO dnc_entries' in sql:
            assert 'ON CONFLICT DO NOTHING' in sql
            if self.entry is not None:
                return 'INSERT 0 0'
            self.entry = self.make_entry(args[2], None)
            return 'INSERT 0 1'
        raise AssertionError(sql)

    async def fetchval(self, sql, *args):
        assert 'SELECT call_attempts FROM leads' in sql
        return self.lead['call_attempts']

    async def fetchrow(self, sql, *args):
        if 'INSERT INTO dnc_entries' in sql:
            self.entry = self.make_entry(args[2], args[5])
            return self.entry
        if "UPDATE leads SET status='dnc'" in sql:
            self.lead.update(status='dnc', last_call_result=args[2])
            return {'id': LEAD}
        if 'FROM leads' in sql and 'AND do_not_call' in sql:
            return {'id': LEAD} if self.lead['do_not_call'] else None
        if 'FROM dnc_entries' in sql:
            active = self.entry and (self.entry['expires_at'] is None or self.entry['expires_at'] > datetime.now(timezone.utc))
            return self.entry if active else None
        raise AssertionError(sql)
